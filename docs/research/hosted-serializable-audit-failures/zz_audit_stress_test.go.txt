// Throwaway load harness for the wayfinder task "reproduce cross-org
// SERIALIZABLE audit failures on the real schema". It drives the real
// revealing read (Values.Get, reveal=true: SERIALIZABLE write with authz reads,
// value reads and a disclosure audit row) across many orgs from several store
// pools on one database, standing in for several server nodes.

func stressEnv(t *testing.T, name string, def int) int {
	t.Helper()
	raw := os.Getenv("HIKYO_AUDIT_STRESS_" + name)
	if raw == "" {
		return def
	}
	v, err := strconv.Atoi(raw)
	if err != nil {
		t.Fatalf("HIKYO_AUDIT_STRESS_%s: %v", name, err)
	}
	return v
}

func TestZZAuditStress(t *testing.T) {
	if os.Getenv("HIKYO_AUDIT_STRESS") == "" {
		t.Skip("HIKYO_AUDIT_STRESS not set")
	}
	orgs, nodes := stressEnv(t, "ORGS", 200), stressEnv(t, "NODES", 3)
	clients, rate := stressEnv(t, "CLIENTS", 48), stressEnv(t, "RATE", 400)
	secs, preload := stressEnv(t, "SECS", 30), stressEnv(t, "PRELOAD_ROWS", 200000)
	ctx := t.Context()

	db1 := seededDB(t, openPostgres)
	keys := keySvc(t, db1)
	values := valueSvc(t, db1)
	for i := range orgs {
		o, p, e := fmt.Sprintf("org_s%d", i), fmt.Sprintf("prj_s%d", i), fmt.Sprintf("env_s%d", i)
		for _, stmt := range []string{
			`INSERT INTO orgs (id, name, active, metadata, created_at) VALUES ('` + o + `', '` + o + `', TRUE, '{}', ` + ts + `)`,
			`INSERT INTO projects (id, org_id, name, created_at) VALUES ('` + p + `', '` + o + `', 'p', ` + ts + `)`,
			`INSERT INTO project_schema_revisions (org_id, project_id, revision) VALUES ('` + o + `', '` + p + `', 0)`,
			`INSERT INTO environments (id, org_id, project_id, name, note, created_at, display_order) VALUES ('` + e + `', '` + o + `', '` + p + `', 'dev', '', ` + ts + `, 0)`,
			`INSERT INTO grants (id, principal_id, capability, org_id, project_id, env_id, created_at)
			 SELECT id || '_` + o + `', principal_id, capability, '` + o + `', NULL, NULL, created_at FROM grants WHERE principal_id = 'usr_custodian' AND org_id = 'org_a'`,
		} {
			execRaw(t, db1, stmt)
		}
		project := domain.Scope{Org: domain.OrgID(o), Project: domain.ProjectID(p)}
		if _, err := keys.Create(ctx, service.LocalPrincipal(custodian), project, service.KeySpec{
			Name: "DB_PASSWORD", Classification: string(schema.Secret),
			Declaration: schema.Declaration{Rule: &schema.Rule{Type: schema.TypeString}},
			Presence:    schema.DefaultPresenceRules(),
		}, nil); err != nil {
			t.Fatalf("create key in %s: %v", o, err)
		}
		if _, _, err := values.Declare(ctx, service.LocalPrincipal(custodian), project, []string{e}, "DB_PASSWORD", "v-"+o); err != nil {
			t.Fatalf("declare value in %s: %v", o, err)
		}
	}
	execRaw(t, db1, fmt.Sprintf(`INSERT INTO audit_tenant_events (id, type, schema_version, occurred_at, occurred_asserted, actor_id, actor_class, scope_class, org_id, outcome, origin, payload)
		SELECT 'pre_' || i, 'value.revealed', 1, now(), true, 'usr_custodian', 'human', 'org', 'org_s' || (i %% %d), 'success', 'api', '{}' FROM generate_series(1, %d) i`, orgs, preload))
	execRaw(t, db1, `VACUUM ANALYZE audit_tenant_events`)
	if path := os.Getenv("HIKYO_AUDIT_STRESS_VARIANT_SQL"); path != "" {
		sql, err := os.ReadFile(path)
		if err != nil {
			t.Fatal(err)
		}
		execRaw(t, db1, string(sql))
	}

	// Further nodes: their own pools on the same admitted database.
	probeKeyringMu.Lock()
	root := bytes.Clone(probeKeyringRegistry[db1].root)
	probeKeyringMu.Unlock()
	dsn := derivedDatabase(t, postgresTestDSN(t), "_isolation")
	svcs := []*service.Values{values}
	for range nodes - 1 {
		db, err := store.Open(ctx, store.Config{Engine: store.EnginePostgres, DSN: dsn}, isolationAdmission(t, db1))
		if err != nil {
			t.Fatal(err)
		}
		t.Cleanup(func() { db.Close() })
		loadAndRegisterKeyring(t, db, bytes.Clone(root))
		svcs = append(svcs, valueSvc(t, db))
	}
	before := queryInt(t, db1, `SELECT count(*) FROM audit_tenant_events`)

	// Open loop: a ticker issues ops at the target rate; an op that finds no
	// idle client within one interval is counted late and dropped.
	type result struct {
		lat time.Duration
		err error
	}
	work := make(chan struct{})
	results := make(chan result, 2*rate*secs+clients) // ticks can exceed rate*secs by a few
	var wg sync.WaitGroup
	for c := range clients {
		svc := svcs[c%len(svcs)]
		wg.Add(1)
		go func() {
			defer wg.Done()
			for range work {
				i := rand.IntN(orgs)
				scope := scopeEnv(domain.OrgID(fmt.Sprintf("org_s%d", i)), domain.ProjectID(fmt.Sprintf("prj_s%d", i)), domain.EnvID(fmt.Sprintf("env_s%d", i)))
				start := time.Now()
				_, err := svc.Get(ctx, service.LocalPrincipal(custodian), scope, "DB_PASSWORD", true)
				results <- result{time.Since(start), err}
			}
		}()
	}
	var late atomic.Int64
	interval := time.Second / time.Duration(rate)
	ticker := time.NewTicker(interval)
	deadline := time.Now().Add(time.Duration(secs) * time.Second)
	for now := range ticker.C {
		if now.After(deadline) {
			break
		}
		select {
		case work <- struct{}{}:
		case <-time.After(interval):
			late.Add(1)
		}
	}
	ticker.Stop()
	close(work)
	wg.Wait()
	close(results)

	var lats []time.Duration
	var ok, serial, other int
	var otherMsgs []string
	for r := range results {
		lats = append(lats, r.lat)
		var pgErr *pgconn.PgError
		switch {
		case r.err == nil:
			ok++
		case errors.As(r.err, &pgErr) && (pgErr.Code == "40001" || pgErr.Code == "40P01"):
			serial++
		default:
			other++
			if len(otherMsgs) < 3 && !slices.Contains(otherMsgs, r.err.Error()) {
				otherMsgs = append(otherMsgs, r.err.Error())
			}
		}
	}
	slices.Sort(lats)
	n := len(lats)
	if n == 0 {
		t.Fatal("no operations ran")
	}
	after := queryInt(t, db1, `SELECT count(*) FROM audit_tenant_events`)
	t.Logf("orgs=%d nodes=%d clients=%d rate=%d secs=%d: attempted=%d ok=%d (%.1f/s) failed_serialization=%d (%.3f%%) failed_other=%d late=%d p50=%v p99=%v audit_rows %d->%d other=%q",
		orgs, nodes, clients, rate, secs, n, ok, float64(ok)/float64(secs), serial, 100*float64(serial)/float64(n), other, late.Load(),
		lats[n/2], lats[n*99/100], before, after, otherMsgs)
}
